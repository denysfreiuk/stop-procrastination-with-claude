# Звички

Кожна клітинка ставиться сама з того, що записують бот і `/journal`. 📵 і 🏋️ можна клацнути й тут: зміниться лише цей рядок у властивостях нотатки дня.

Цілі й сам список звичок — угорі скрипта нижче. Потрібен Dataview з увімкненою опцією **Enable JavaScript Queries**.

```dataviewjs
// Goals and the habits themselves: each habit is a column, remove a line to hide it.
// The bot's pinned dashboard reads GOALS from this very line, so keep it on one line.
const GOALS = { focusMinutes: 50, sleepHours: 7 };
const MONTHS_SHOWN = 3;

const HABITS = [
  {
    icon: "📵",
    name: "Без тіктока",
    done: (page) => page.tiktok_relapse !== true,
    // A relapse is final: today's relapse ends the streak at once.
    failsToday: true,
    property: { key: "tiktok_relapse", value: (checked) => !checked },
  },
  {
    icon: "🎯",
    name: `Фокус від ${GOALS.focusMinutes} хв (/f)`,
    done: (page) => number(page.focus_minutes) >= GOALS.focusMinutes,
  },
  { icon: "📚", name: "Запис у «Що вивчив» (/l)", done: (page) => hasLearned(page) },
  { icon: "🔤", name: "Нові слова (/w)", done: (page) => number(page.new_words) > 0 },
  {
    icon: "🏋️",
    name: "Зал (/gym)",
    done: (page) => page.gym === true,
    property: { key: "gym", value: (checked) => checked },
  },
  {
    icon: "😴",
    name: `Сон від ${GOALS.sleepHours} год (/sleep)`,
    done: (page) => isFilled(page.sleep_hours) && number(page.sleep_hours) >= GOALS.sleepHours,
  },
  { icon: "📝", name: "Підсумок дня (/journal)", done: (page) => hasSummary(page) },
];

const DAILY = "Journal/Daily";
const LEARNED_SECTION = "Що вивчив";
const SUMMARY_SECTION = "Підсумок";
const MONTH_NAMES = [
  "Січень", "Лютий", "Березень", "Квітень", "Травень", "Червень",
  "Липень", "Серпень", "Вересень", "Жовтень", "Листопад", "Грудень",
];
const WEEKDAYS = ["Нд", "Пн", "Вт", "Ср", "Чт", "Пт", "Сб"];
const DAY = 24 * 60 * 60 * 1000;

const number = (value) => Number(value) || 0;
const isFilled = (value) => value !== null && value !== undefined && value !== "";
// Days are UTC midnights, so adding a day never trips over daylight saving time.
const timeOf = (name) => Date.parse(`${name}T00:00:00Z`);
const nameOf = (time) => new Date(time).toISOString().slice(0, 10);

const byDay = new Map(dv.pages(`"${DAILY}"`).array().map((page) => [page.file.name, page]));
const today = timeOf(dv.date("today").toFormat("yyyy-LL-dd"));
const firstDay = Math.min(today, ...[...byDay.keys()].map(timeOf).filter(Number.isFinite));

function hasLearned(page) {
  return page.file.lists.array().some((item) => item.section?.subpath === LEARNED_SECTION);
}

// The summary counts once anything is written under its heading.
function hasSummary(page) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  const cache = file && app.metadataCache.getFileCache(file);
  const headings = cache?.headings ?? [];
  const index = headings.findIndex((heading) => heading.heading === SUMMARY_SECTION);
  if (index < 0) return false;
  const start = headings[index].position.start.line;
  const next = headings.slice(index + 1).find((heading) => heading.level <= headings[index].level);
  const end = next ? next.position.start.line : Infinity;
  return (cache.sections ?? []).some((section) => {
    const line = section.position.start.line;
    return section.type !== "heading" && line > start && line < end;
  });
}

// Today is still going: a habit not done yet keeps the streak, a day without a note breaks it.
function streak(habit) {
  const todayPage = byDay.get(nameOf(today));
  if (todayPage && habit.failsToday && !habit.done(todayPage)) return 0;
  let days = todayPage && habit.done(todayPage) ? 1 : 0;
  for (let time = today - DAY; ; time -= DAY) {
    const page = byDay.get(nameOf(time));
    if (!page || !habit.done(page)) return days;
    days++;
  }
}

// Only the property's own line changes, the way the bot writes, so git merges it with the bot's writes.
function withProperty(text, key, value) {
  const lines = text.split("\n");
  if (lines[0].trim() !== "---") return text;
  const end = lines.findIndex((line, index) => index > 0 && line.trim() === "---");
  if (end < 0) return text;
  const line = `${key}: ${value}${lines[0].endsWith("\r") ? "\r" : ""}`;
  const at = lines.findIndex((existing, index) => index > 0 && index < end && existing.startsWith(`${key}:`));
  if (at > 0) lines[at] = line;
  else lines.splice(end, 0, line);
  return lines.join("\n");
}

async function save(page, habit, box) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  try {
    await app.vault.process(file, (text) => withProperty(text, habit.property.key, habit.property.value(box.checked)));
  } catch (error) {
    box.checked = !box.checked;
    console.error("Звички: не вдалося зберегти", error);
  }
}

function make(tag, parent, text) {
  const element = parent.appendChild(parent.ownerDocument.createElement(tag));
  if (text !== undefined) element.textContent = text;
  return element;
}

function checkbox(cell, page, habit) {
  const box = make("input", cell);
  box.type = "checkbox";
  box.className = "task-list-item-checkbox";
  box.checked = habit.done(page);
  box.title = habit.name;
  if (habit.property) {
    box.addEventListener("change", () => save(page, habit, box));
  } else {
    box.title += " — ставиться сама із записів";
    box.addEventListener("click", (event) => event.preventDefault());
  }
}

function monthTable(days) {
  const date = new Date(days[0]);
  make("h2", dv.container, `${MONTH_NAMES[date.getUTCMonth()]} ${date.getUTCFullYear()}`);
  const table = make("table", dv.container);
  table.className = "habits";
  const head = make("tr", make("thead", table));
  make("th", head, "День");
  for (const habit of HABITS) make("th", head, habit.icon).title = habit.name;

  const body = make("tbody", table);
  for (const time of days) {
    const day = new Date(time);
    const page = byDay.get(nameOf(time));
    const row = make("tr", body);
    const label = `${day.getUTCDate()} ${WEEKDAYS[day.getUTCDay()]}`;
    const dayCell = make("td", row);
    if (page) dv.el("span", dv.fileLink(page.file.path, false, label), { container: dayCell });
    else dayCell.textContent = label;
    // A day without a note has nothing to tick, and it breaks every streak.
    for (const habit of HABITS) {
      const cell = make("td", row);
      if (page) checkbox(cell, page, habit);
    }
  }

  const total = make("tr", make("tfoot", table));
  make("td", total, "Разом");
  for (const habit of HABITS) {
    const done = days.filter((time) => byDay.has(nameOf(time)) && habit.done(byDay.get(nameOf(time)))).length;
    make("td", total, `${done}/${days.length}`).title = habit.name;
  }
}

const style = make("style", dv.container);
style.textContent =
  ".habits td, .habits th { text-align: center; padding: 2px 6px; }" +
  ".habits td:first-child, .habits th:first-child { text-align: left; white-space: nowrap; }" +
  ".habits input { margin: 0; }" +
  ".habits tfoot td { font-size: 0.8em; opacity: 0.8; }";

make("p", dv.container, HABITS.map((habit) => `${habit.icon} ${habit.name}`).join(" · "));
make("p", dv.container, `Поспіль зараз: ${HABITS.map((habit) => `${habit.icon} ${streak(habit)}`).join(" · ")}`);

// Months newest first, days inside a month in order, the way a paper tracker reads.
const months = [];
for (let time = today; time >= firstDay; time -= DAY) {
  const month = nameOf(time).slice(0, 7);
  if (months.at(-1)?.month !== month) {
    if (months.length === MONTHS_SHOWN) break;
    months.push({ month, days: [] });
  }
  months.at(-1).days.unshift(time);
}
for (const { days } of months) monthTable(days);
```
